cmake_minimum_required(VERSION 3.24)
project(aftermath_decoder LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "First supported target is Windows x64; no Linux fixture has been validated")
endif()
set(AFTERMATH_SDK_PATH "$ENV{AFTERMATH_SDK_PATH}" CACHE PATH "User-installed official NVIDIA Aftermath SDK")
find_path(AFTERMATH_INCLUDE GFSDK_Aftermath_GpuCrashDumpDecoding.h PATHS "${AFTERMATH_SDK_PATH}/include" NO_DEFAULT_PATH REQUIRED)
find_library(AFTERMATH_LIBRARY NAMES GFSDK_Aftermath_Lib.x64 PATHS "${AFTERMATH_SDK_PATH}/lib/x64" "${AFTERMATH_SDK_PATH}/lib" NO_DEFAULT_PATH REQUIRED)
find_file(AFTERMATH_RUNTIME GFSDK_Aftermath_Lib.x64.dll PATHS "${AFTERMATH_SDK_PATH}/lib/x64" "${AFTERMATH_SDK_PATH}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(aftermath-decoder main.cpp)
target_compile_features(aftermath-decoder PRIVATE cxx_std_17)
target_include_directories(aftermath-decoder PRIVATE "${AFTERMATH_INCLUDE}")
target_link_libraries(aftermath-decoder PRIVATE "${AFTERMATH_LIBRARY}")
# No SDK DLL is copied or redistributed. Runtime must be on the user's PATH.
